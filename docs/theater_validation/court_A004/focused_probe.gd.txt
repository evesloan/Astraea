# Scratch-only focused test. Main scene must construct Court A itself.
# Reuse existing exact-old-theater and physical loaded-cart checks; no test builder.
var output := ""

var screen: Dictionary
var court_data: Dictionary

var walking_peak := 0.0
var walking_last := Vector3.ZERO

func same_faces(actual: PackedVector3Array, expected: PackedVector3Array) -> void:
	assert(not actual.is_empty() and actual.to_byte_array()==expected.to_byte_array(),"Exact raw source triangles must survive into actual-world physics storage")
	verified_contact_points+=actual.size()

func inspect_court() -> void:
	court_data=JSON.parse_string(FileAccess.get_file_as_string("res://data/architecture/COURT_A_RUNTIME_INCREMENT004.json"))
	court=probe.ship.get_node("ShipSpace/Districts/"+str(court_data.root_name)) as Node3D
	assert(court!=null and court.get_parent()==probe.venue.get_parent() and not probe.venue.is_ancestor_of(court))
	assert(court.transform==Transform3D(Basis.IDENTITY,probe.xyz(court_data.court_origin_native)))
	assert(court.get_meta("source_sha256")==court_data.source_sha256 and court.get_meta("lights_added")==0)
	assert(court.find_children("*","Light3D",true,false).is_empty())
	var builder=load("res://scripts/court_a_increment004.gd")
	var expected_bodies: Dictionary={}
	for batch in court_data.batches:
		if batch.kind!="authored":continue
		var mesh:=court.get_node(str(batch.name)) as MeshInstance3D
		assert(mesh!=null and mesh.mesh!=null)
		probe.assert_transform(mesh.transform,builder.transform(batch.native_local_matrix),str(batch.name))
		assert(mesh.get_meta("source_semantic_ids")==batch.semantic_ids)
		if batch.collides:
			var body:=mesh.get_node("CourtContact_"+str(batch.name)) as StaticBody3D
			expected_bodies[body]=true
			assert(body.transform==Transform3D.IDENTITY)
			same_faces((body.get_child(0).shape as ConcavePolygonShape3D).get_faces(),builder.raw_faces(mesh.mesh))
		else:assert(mesh.find_children("*","CollisionObject3D",true,false).is_empty())
	var template:=builder.load_template(court_data.files.prototype) as Node3D
	var definitions: Dictionary={}
	for batch in court_data.batches:
		if batch.kind!="prototype" or not batch.collides:continue
		var part:=template.find_child(str(batch.name),true,false) as MeshInstance3D
		var identity:=str(batch.definition)
		if not definitions.has(identity):definitions[identity]=PackedVector3Array()
		var faces: PackedVector3Array=definitions[identity]
		faces.append_array(builder.raw_faces(part.mesh,builder.relative_transform(part,template)))
		definitions[identity]=faces
	var placements: Array=[]
	for record in court_data.instances:
		placements.append(str(record.id))
		if not record.collides:continue
		var body:=court.get_node("CourtContact_"+str(record.id)) as StaticBody3D
		expected_bodies[body]=true
		probe.assert_transform(body.global_transform,builder.transform(record.native_matrix),str(record.id))
		same_faces((body.get_child(0).shape as ConcavePolygonShape3D).get_faces(),definitions[str(record.definition)])
	template.free()
	assert(court.get_meta("source_instance_ids")==placements)
	var installed:=court.find_children("*","StaticBody3D",true,false)
	assert(installed.size()==expected_bodies.size() and installed.size()==101)
	for body in installed:
		assert(expected_bodies.has(body) and body.collision_layer==1 and body.collision_mask==1 and body.get_child_count()==1)
		var shape:=body.get_child(0) as CollisionShape3D
		assert(shape!=null and not shape.disabled and shape.transform==Transform3D.IDENTITY)
		assert(shape.shape is ConcavePolygonShape3D and absf(shape.shape.margin-.001)<.000001)
	var ramps: Node=probe.ship.get_node("ShipSpace/Districts/GrandLanternTheater_StairWalkingContacts")
	assert(ramps.get_child_count()==8 and ramps.find_children("*","GeometryInstance3D",true,false).is_empty())

func check_stance(point: Vector3, label: String) -> void:
	probe.floor_at(point,"D11_theater_flat_00")
	var capsule:=CapsuleShape3D.new()
	capsule.radius=float(geometry.routes.capsule_radius_m)
	capsule.height=float(geometry.routes.capsule_height_m)
	sweep_probe.clear_shape(probe.space,capsule,Transform3D(Basis.IDENTITY,point+Vector3.UP*.91),probe.exclusions(),label)

func ordinary_walk(target: Vector3) -> void:
	var actor: CharacterBody3D=probe.actor
	var initial:=Vector2(target.x-actor.global_position.x,target.z-actor.global_position.z).length()
	var reached:=false
	for frame in range(ceili(initial/3.1*60)+360):
		var delta:=Vector3(target.x-actor.global_position.x,0,target.z-actor.global_position.z)
		var speed:=Vector2(actor.velocity.x,actor.velocity.z).length()
		if delta.length()<.075 and speed<.08:reached=true;break
		actor.rotation.y=atan2(-delta.x,-delta.z)
		key(KEY_W,delta.length()>.06+speed*speed/32.0)
		await physics_frame
		walking_frames+=1
		var step:=actor.global_position.distance_to(walking_last)
		assert(step<.08 and actor.global_position.is_finite() and absf(actor.global_position.y-FLOOR)<.14)
		walking_distance+=step;walking_last=actor.global_position
		walking_peak=maxf(walking_peak,Vector2(actor.velocity.x,actor.velocity.z).length())
		probe.floor_at(actor.global_position)
	key(KEY_W,false)
	assert(reached,"Court A ordinary W walk stalled at %s toward %s"%[actor.global_position,target])
	probe.check_lights("Court A walk "+str(target))
