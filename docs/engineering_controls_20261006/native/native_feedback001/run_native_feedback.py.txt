def head():return subprocess.check_output(['git','-C',str(G),'rev-parse','HEAD'],text=True).strip()

 receipt=R/'controls_run002/CONTROLS_RECEIPT.json';validation=R/'controls_run002/VALIDATION.json';assert sha(receipt)==fixed['all120_receipt_sha256'] and sha(validation)==fixed['all120_validation_sha256']

 # not proof the other namespace is idle. Never inspect/kill runner.json's PID.

  try:name=(proc/'comm').read_text().strip().lower()

  native=json.loads((O/'NATIVE_RECEIPT.json').read_text());assert native['status']=='PASS' and len(native['views'])==4 and len(native['cue_ranges'])==2 and native['capture_effect_removed']

  path=O/'application_mix.wav';assert sha(path)==native['mixed_output']['wav_sha256']
