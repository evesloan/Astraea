def add(name, verts, faces, role, owner, contact=True):
    assert name not in [q['name']for q in parts]
    assert owner in owners
    parts.append(dict(name=name,verts=verts,faces=faces,role=role,owner=owner,contact=contact))

def extrude(poly, lo, hi, name, role, owner, contact=True):
    from shapely.geometry.polygon import orient
    vs=[];fs=[]
    for g in ([poly]if poly.geom_type=='Polygon'else poly.geoms):
        if g.area<1e-8:continue
        g=orient(g,sign=1)
        for t in constrained_delaunay_triangles(g).geoms:
            cs=list(t.exterior.coords)[:3]
            if not t.exterior.is_ccw:cs.reverse()
            b=len(vs);vs += [[x,lo,z]for x,z in cs]+[[x,hi,z]for x,z in cs]
            fs += [[b,b+1,b+2],[b+5,b+4,b+3]]
        for ring in [g.exterior,*g.interiors]:
            for (x,z),(xx,zz)in zip(ring.coords,list(ring.coords)[1:]):
                b=len(vs);vs += [[x,lo,z],[xx,lo,zz],[xx,hi,zz],[x,hi,z]];fs.append([b+3,b+2,b+1,b])
    add(name,vs,fs,role,owner,contact)

def box(name,center,size,role,owner,contact=True):
    x,y,z=center;sx,sy,sz=size
    extrude(rectangle(x-sx/2,z-sz/2,x+sx/2,z+sz/2),y-sy/2,y+sy/2,name,role,owner,contact)

def top(row):
    vs=row['verts'];y=max(v[1]for v in vs)
    ps=[Polygon([(vs[i][0],vs[i][2])for i in f])for f in row['faces']if all(abs(vs[i][1]-y)<1e-7 for i in f)]
    return unary_union([p for p in ps if p.area>1e-8]),y

# Each bay follows the real opaque wall. Positive depth points inside the court.
slope=(100-63)/(491.123-423.164)
def edge(z):return 100 if z<=423.164 else 100-(z-423.164)*slope
def wall_frame(z0,z1):
    a=(edge(z0),z0);b=(edge(z1),z1);length=math.dist(a,b);t=((b[0]-a[0])/length,(b[1]-a[1])/length);n=(-t[1],t[0])
    return a,t,n,length
def wall_solid(name,z0,z1,outline,depth0,depth1,role,owner):
    a,t,n,length=wall_frame(z0,z1)
    poly=Polygon(outline);assert poly.is_valid and poly.area>0
    # Map an upright local polygon (distance along wall, height) into the court.
    local=[];faces=[]
    for tri in constrained_delaunay_triangles(poly).geoms:
        cs=list(tri.exterior.coords)[:3]
        if not tri.exterior.is_ccw:cs.reverse()
        k=len(local);local += [(u,y,depth0)for u,y in cs]+[(u,y,depth1)for u,y in cs]
        faces += [[k,k+1,k+2],[k+5,k+4,k+3]]
    cs=list(poly.exterior.coords)
    for (u,y),(v,h)in zip(cs,cs[1:]):
        k=len(local);local += [(u,y,depth0),(v,h,depth0),(v,h,depth1),(u,y,depth1)];faces.append([k+3,k+2,k+1,k])
    verts=[[a[0]+t[0]*u+n[0]*depth,y,a[1]+t[1]*u+n[1]*depth]for u,y,depth in local]
    add(name,verts,faces,role,owner)
