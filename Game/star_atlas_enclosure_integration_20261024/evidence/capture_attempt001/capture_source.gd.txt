func _initialize()->void:call_deferred("run")
func run()->void:
	var main:Node=load("res://scenes/main_retained_enclosure_integrated_20261024.tscn").instantiate()
	root.add_child(main)
	var capture_only:bool="--capture" in OS.get_cmdline_user_args()
	if capture_only:main.get_node("World360").render_target_update_mode=SubViewport.UPDATE_DISABLED
	var ship:Node3D=main.get_node("World360/Astraea")
	while not ship.get_meta("runtime_ready",false):await process_frame
	for i in range(8):await physics_frame
	paused=true
	if capture_only:
		await capture_current_ship(main,ship)
		quit();return
	var districts:Node3D=ship.get_node("ShipSpace/Districts")
	var upper:Node3D=districts.get_node("D15_D17_StarAtlasUpper_WIP")
	assert(upper.get_meta("source_part_count")==514 and upper.get_meta("D15_cover_reused") and upper.get_meta("new_guest_units")==0)
	var enclosure:Node3D=upper.get_node("StarAtlasEnclosure20261022")
	var addition_meshes:Array=enclosure.find_children("*","MeshInstance3D",true,false)
	assert(addition_meshes.size()==140 and enclosure.get_meta("new_capacity")==0)
	assert(upper.get_meta("star_atlas_living_commons_20261022").shared_asset_placements==98)
	assert(enclosure.find_children("*","Light3D",true,false).is_empty())
	assert(main.get_node("World360").use_occlusion_culling and main.get_node("World360").audio_listener_enable_3d)
	# Production uses unsnapped surface arrays. Mesh.get_faces() constructs a
	# TriangleMesh that snaps coordinates to 0.0001m in Godot4.6.3.
	# Compare identical representations, keeping exact equality (no tolerance).
	var witness:=PackedVector3Array();var cell_key:=Vector3i.ZERO
	var witness_mesh:="";var snap_delta:=0.0
	for mesh:MeshInstance3D in addition_meshes:
		var xf:Transform3D=ship.global_transform.affine_inverse()*mesh.global_transform
		for surface in range(mesh.mesh.get_surface_count()):
			var material:=mesh.get_active_material(surface)as BaseMaterial3D
			if material==null or material.transparency!=BaseMaterial3D.TRANSPARENCY_DISABLED or material.albedo_color.a<.999:continue
			var arrays:Array=mesh.mesh.surface_get_arrays(surface)
			var vertices:PackedVector3Array=arrays[Mesh.ARRAY_VERTEX]
			var indices:PackedInt32Array=arrays[Mesh.ARRAY_INDEX]if arrays[Mesh.ARRAY_INDEX]!=null else PackedInt32Array()
			var count:int=indices.size()if not indices.is_empty()else vertices.size()
			for i in range(0,count-2,3):
				var a:Vector3=xf*vertices[indices[i]if not indices.is_empty()else i]
				var b:Vector3=xf*vertices[indices[i+1]if not indices.is_empty()else i+1]
				var c:Vector3=xf*vertices[indices[i+2]if not indices.is_empty()else i+2]
				if (b-a).cross(c-a).length()*.5>=2.0:
					witness=PackedVector3Array([a,b,c]);witness_mesh=str(mesh.get_path())
					var center:Vector3=(a+b+c)/3.0
					cell_key=Vector3i(floori(center.x/48),floori(center.y/16),floori(center.z/48))
					for v in witness:snap_delta=maxf(snap_delta,v.distance_to(v.snappedf(.0001)))
					break
			if not witness.is_empty():break
		if not witness.is_empty():break
	var registered:=false
	if not witness.is_empty():
		var cell:=ship.get_node_or_null("VerifiedStaticOpaqueFaceOccluders/Cell_%s_%s_%s"%[cell_key.x,cell_key.y,cell_key.z])as OccluderInstance3D
		if cell!=null:
			var vertices:PackedVector3Array=cell.occluder.get_vertices()
			for i in range(0,vertices.size(),3):
				if vertices[i]==witness[0] and vertices[i+1]==witness[1] and vertices[i+2]==witness[2]:registered=true;break
	print("STAR_ATLAS_OCCLUDER_WITNESS ",JSON.stringify({"registered":registered,"mesh":witness_mesh,"cell":str(cell_key),"get_faces_snap_delta_m":snap_delta}))
	var lower:Node3D=districts.get_node("D12_D14_StarAtlas_WIP")
	assert(lower.get_node("StarAtlasPassengerLift").floor_heights==[78.0,84.0,88.5])
	var home_root:Node3D=districts.get_node("StarAtlasRoomCandidates")
	var retained_homes:=0
	for entry in ship.get_node("ShipSystems/ReceivedRoomResidency").entries:
		if entry.parent==home_root:
			assert(is_instance_valid(entry.room))
			retained_homes+=1
	assert(retained_homes==85,"Count registered homes, including temporarily detached residents")
	var d:Dictionary=JSON.parse_string(FileAccess.get_file_as_string("res://data/architecture/STAR_ATLAS_UPPER_COMMONS.json"))
	for path in d.source_pins:assert(FileAccess.get_sha256("res://"+str(path).trim_prefix("Game/"))==d.source_pins[path])
	var excluded:Array[RID]=[]
	for body in ship.find_children("*","CharacterBody3D",true,false):excluded.append(body.get_rid())
	for body in ship.find_children("*","RigidBody3D",true,false):excluded.append(body.get_rid())
	var physics:=ship.get_world_3d().direct_space_state
	var points:Array[Dictionary]=[]
	for y in [93.0,97.5,102.0]:
		for x in [-65.5,-61.0,-55.0,-50.0]:points.append({"p":Vector3(x,y,0),"stair":false})
		points.append({"p":Vector3(-76,y,-8),"stair":false})
	for z in [-15.0,-24.0,-31.0,15.0,24.0,29.0]:points.append({"p":Vector3(-84,97.5,z),"stair":false})
	for y in [93.0,97.5]:
		points.append({"p":Vector3(-70.8,y+1.125,6.54),"stair":true})
		points.append({"p":Vector3(-69.3,y+2.25,10.2),"stair":false})
		points.append({"p":Vector3(-67.8,y+3.375,6.54),"stair":true})
	var errors:Array=[]
	if not registered:errors.append({"kind":"occluder_registration","mesh":witness_mesh,"cell":str(cell_key)})
	for item in points:
		var p:Vector3=item.p
		var ray:=PhysicsRayQueryParameters3D.create(p+Vector3.UP*.2,p-Vector3.UP*.35);ray.exclude=excluded
		var hit:=physics.intersect_ray(ray)
		if hit.is_empty() or absf(hit.position.y-p.y)>.035:errors.append({"kind":"floor","point":str(p),"hit":str(hit.get("position","none"))})
		var query:=PhysicsShapeQueryParameters3D.new();var capsule:=CapsuleShape3D.new();capsule.radius=.3;capsule.height=1.8
		query.shape=capsule;query.exclude=excluded;query.margin=.001;query.transform.origin=p+Vector3.UP*(1.0 if item.stair else .925)
		var collisions:=physics.intersect_shape(query,16)
		if not collisions.is_empty():
			var names:Array=[]
			for col in collisions:names.append(str(col.collider.get_path()))
			errors.append({"kind":"clearance","point":str(p),"colliders":names})
	var report={"status":"PASS" if errors.is_empty() else "FAIL","parts":514,"retained_enclosure_parts":140,"preserved_living_placements":98,"new_opaque_triangle_registered":registered,"witness_mesh":witness_mesh,"witness_get_faces_snap_delta_m":snap_delta,"viewport_occlusion_enabled":true,"render_or_simulation_savings_measured":false,"new_guest_units":0,"retained_homes":85,"lower_lift_stops":3,"source_pins":d.source_pins.size(),"floor_and_clearance_points":points.size(),"errors":errors,"scope":"Actual main installation, retained source pins and bounded public-ring/home-link/stair witnesses. Original27 public-ring/home-link/stair witnesses unchanged. No new standalone fixture; actual integrated presentation and all runtime construction. Detailed traversal, final visual review and service completion remain unfinished."}
	FileAccess.open("res://star_atlas_enclosure_integration_20261024/evidence/INSTALLED_CHECK.json",FileAccess.WRITE).store_string(JSON.stringify(report,"  "));print("STAR_ATLAS_RETAINED_ENCLOSURE ",JSON.stringify(report));quit(0 if errors.is_empty() else 1)

	var budget:Node=ship.get_node("ShipSystems/LocalShadowSourceBudget")
	budget.refresh()

	view.render_target_update_mode=SubViewport.UPDATE_ONCE
	await process_frame;await RenderingServer.frame_post_draw
	# A second frame allows queued shader/light updates; no exposure alteration.
	view.render_target_update_mode=SubViewport.UPDATE_ONCE
	await process_frame;await RenderingServer.frame_post_draw
	var image_path:="res://star_atlas_enclosure_integration_20261024/evidence/STAR_ATLAS_INSTALLED_CONTEXT_UNFINISHED.png"
