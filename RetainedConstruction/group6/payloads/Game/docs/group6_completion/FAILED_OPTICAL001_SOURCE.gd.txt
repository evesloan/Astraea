func optical_checks(host:Node3D,annex:Node3D,addition:Node3D,actor:CharacterBody3D,budget:Node)->void:

	var space:=host.get_world_3d().direct_space_state;var sources:Array=[];var fixtures:Array=[]

		var target:=vec(spec.target_xyz);var q:=PhysicsRayQueryParameters3D.create(light.global_position,target);q.exclude=[actor.get_rid()];var hit:=space.intersect_ray(q)
		record(str(id)+"_exact_mounted_source",light.global_position.distance_to(vec(spec.source_xyz))<.00001)
		record(str(id)+"_real_receiver_path",hit.is_empty())
		record(str(id)+"_within_cone_and_reach",light.global_position.distance_to(target)<light.spot_range and (-light.global_basis.z).dot((target-light.global_position).normalized())>.999)
		var surface_hit:Dictionary={}
		if spec.receiver_kind=="retained_thermal_water_visual":
			record(str(id)+"_protected_water_height",absf(target.y-45.08)<.00001 and target.x==82.6 and target.z in [211.5,223.5])
		else:
			q=PhysicsRayQueryParameters3D.create(target,target+(Vector3.RIGHT*.12 if spec.receiver_kind=="mechanical_isolator_face" else Vector3.DOWN*.12));q.exclude=[actor.get_rid()];surface_hit=space.intersect_ray(q)
			record(str(id)+"_real_receiver_surface",not surface_hit.is_empty())
		fixtures.append({"id":id,"receiver_kind":spec.receiver_kind,"clear_path":hit.is_empty(),"target_xyz":spec.target_xyz,"surface":str(surface_hit.collider.get_path()) if not surface_hit.is_empty() else "protected visual thermal water face","range_m":light.spot_range,"energy":light.light_energy})
	var viewpoints:Array=[]
	for spec in [["dry_arrival",Vector3(60.6,44.03,240)],["south_boarding",Vector3(76,45.48,239.6)],["west_rest",Vector3(64.8,45.48,220)],["rinse_inside",Vector3(59.4,45.48,231)],["quiet_court",Vector3(76.2,45.48,217)],["water_care",Vector3(96.8,45.48,210.6)],["launch",Vector3(72,49.38,194.4)]]:
		actor.global_position=spec[1]
		for i in range(2):await physics_frame
		budget.refresh();var active:Array=[];var reaches_actor:Array=[]
		for light:Light3D in budget.sources:
			if not is_instance_valid(light) or not light.is_inside_tree() or not budget.requested_shadow.get(light.get_instance_id(),false):continue
			record(str(spec[0])+"_shadow_retained_"+str(light.name),light.visible==light.shadow_enabled)
			if not light.visible:continue
			active.append(str(light.name));record(str(spec[0])+"_actual_visibility_"+str(light.name),budget._has_visible_sample(light))
			var receiver:Vector3=actor.global_position+Vector3.UP*.8;var offset:=receiver-light.global_position
			var reaches:=offset.length()<(light.spot_range if light is SpotLight3D else light.omni_range)
			if light is SpotLight3D:reaches=reaches and offset.normalized().dot(-light.global_basis.z)>=cos(deg_to_rad(light.spot_angle))
			if reaches and budget._clear_sample(light.global_position,receiver):reaches_actor.append(str(light.name))
		record(str(spec[0])+"_eight_source_cap",active.size()>0 and active.size()<=8)
		record(str(spec[0])+"_light_reaches_real_body_sample",not reaches_actor.is_empty())
		viewpoints.append({"fixture":spec[0],"active_sources":active,"body_sample_reached_by":reaches_actor,"scope":"Independent stationary optical fixtures; no walking or luminance/target-hardware claim"})
	optical_evidence={"fixture_count":fixtures.size(),"fixtures":fixtures,"viewpoints":viewpoints}
