def axes_unique(a):
    if not len(a):return np.empty((0,3))
    a=np.array(a,float);a=a[np.linalg.norm(a,axis=1)>1e-8];a/=np.linalg.norm(a,axis=1)[:,None]
    for row in a:
        first=np.flatnonzero(np.abs(row)>1e-7)[0]
        if row[first]<0:row*=-1
    return np.unique(np.round(a,7),axis=0)
def components(tri):
    # Weld only actual shared edges. Normals/UV seam duplication cannot split
    # a solid face into an unsafe artificial surface-only component.
    quant=np.round(tri.reshape(-1,3)/.00002).astype(np.int64);_,inv=np.unique(quant,axis=0,return_inverse=True);indices=inv.reshape(-1,3)
    parent=list(range(len(tri)));edges={}
    def find(i):
        while parent[i]!=i:parent[i]=parent[parent[i]];i=parent[i]
        return i
    for i,row in enumerate(indices):
        for a,b in [(row[0],row[1]),(row[1],row[2]),(row[2],row[0])]:
            key=tuple(sorted((int(a),int(b))))
            if key in edges:parent[find(i)]=find(edges[key])
            else:edges[key]=i
    groups=collections.defaultdict(list)
    for i in range(len(tri)):groups[find(i)].append(i)
    out=[]
    for ids in groups.values():
        t=tri[ids];verts=np.unique(t.reshape(-1,3),axis=0).astype(float);center=verts.mean(axis=0);edge_counts=collections.Counter(tuple(sorted((int(a),int(b))))for i in ids for a,b in [(indices[i,0],indices[i,1]),(indices[i,1],indices[i,2]),(indices[i,2],indices[i,0])])
        closed=all(v==2 for v in edge_counts.values());n=np.cross(t[:,1]-t[:,0],t[:,2]-t[:,0]);length=np.linalg.norm(n,axis=1);n=n[length>1e-9]/length[length>1e-9,None];points=t[length>1e-9,0];n[np.einsum('ij,ij->i',n,center-points)>0]*=-1
        convex=closed and bool(len(n)) and len(verts)<=256 and bool(np.max(n@verts.T-np.einsum('ij,ij->i',n,points)[:,None])<.0001)
        dirs=np.concatenate([t[:,1]-t[:,0],t[:,2]-t[:,1],t[:,0]-t[:,2]])
        out.append({'lo':verts.min(axis=0),'hi':verts.max(axis=0),'verts':verts,'normals':axes_unique(n),'edges':axes_unique(dirs),'closed':closed,'convex':convex,'triangles':len(t)})
    return out
def prism_axes(v):
    faces=[[2,1,0],[3,4,5],[0,1,4,3],[1,2,5,4],[2,0,3,5]];normals=[];edges=[]
    for face in faces:
        a,b,c=v[face[:3]];normals.append(np.cross(b-a,c-a))
        for i,j in zip(face,face[1:]+face[:1]):edges.append(v[j]-v[i])
    return axes_unique(normals),axes_unique(edges)

    # A 0.5mm clear separation is required; touching receivers are held for
    # an explicit collar/owner contract, not silently overlaid.
