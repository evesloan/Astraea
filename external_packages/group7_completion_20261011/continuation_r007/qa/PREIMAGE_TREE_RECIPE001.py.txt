"""Original medium conservatory trees: roots, tapered trunk, boughs, shoots, blades.

No retained tree is stretched, hidden or overwritten. This explicit editable
source hierarchy is scripted original construction, not biological acceptance.
"""
from pathlib import Path
import json, math
import numpy as np
P=Path(__file__).resolve().parents[1]
plan=json.loads((P/'docs/BOTANICAL_PLAN.json').read_text())
data=json.loads((P/'docs/ARCHITECTURE_PARTS.json').read_text());parts=data['parts'];attachments=[]
assert not any(q.get('new_tree_source')for q in parts),'Run architecture preparer before a new tree revision'
def volume(v,faces):
    v=np.array(v,float);v-=v.mean(0);total=0.
    for f in faces:
        for i in range(1,len(f)-1):total+=np.dot(v[f[0]],np.cross(v[f[i]],v[f[i+1]]))/6
    return abs(float(total))
for tree in plan['procedural_trees']:
    base=np.array(tree['position'],float);h=tree['height'];yaw=tree['yaw'];name=tree['id'];cell=tree['cell']
    rot=np.array([[math.cos(yaw),0,math.sin(yaw)],[0,1,0],[-math.sin(yaw),0,math.cos(yaw)]])
    def world(q):return (base+rot@np.array(q,float)).tolist()
    def tube(suffix,path,radii,parent=None,parent_point=None,sides=10):
        p=np.array(path,float);verts=[]
        for i,q in enumerate(p):
            tangent=p[min(i+1,len(p)-1)]-p[max(0,i-1)];tangent/=np.linalg.norm(tangent)
            helper=np.eye(3)[int(np.argmin(np.abs(tangent)))];u=np.cross(tangent,helper);u/=np.linalg.norm(u);v=np.cross(tangent,u)
            verts.extend(world(q+radii[i]*(u*math.cos(k*math.tau/sides)+v*math.sin(k*math.tau/sides)))for k in range(sides))
        faces=[list(reversed(range(sides))),list(range((len(p)-1)*sides,len(p)*sides))]
        for i in range(len(p)-1):
            for k in range(sides):faces.append([i*sides+k,i*sides+(k+1)%sides,(i+1)*sides+(k+1)%sides,(i+1)*sides+k])
        v=volume(verts,faces);assert v>1e-9,suffix
        part={'name':name+'_'+suffix,'vertices_world_xyz':verts,'faces':faces,'role':'bark','category':'solid_tree_bark','cell':cell,'volume_m3':v,'new_tree_source':True,'editable_control_path_local_xyz':path,'control_radii_m':radii,'parent_path':parent}
        parts.append(part)
        if parent is not None:attachments.append({'child':part['name'],'parent':parent,'shared_or_embedded_joint_world_xyz':world(parent_point if parent_point is not None else path[0]),'construction_joint_only':True})
        return part['name']
    trunk_points=[[0,0,0],[.05,h*.25,.04],[-.12,h*.49,.10],[.10,h*.72,.12],[.20,h*.87,.02]]
    trunk=tube('Trunk',trunk_points,[.27,.23,.17,.105,.035],sides=12)
    for k,angle in enumerate([-.8,1.35,2.9,4.35]):
        root=[.94*math.cos(angle),.025,.94*math.sin(angle)]
        tube('RootFlare_'+str(k),[root,[.38*math.cos(angle),.075,.38*math.sin(angle)],[0,.12,0]],[.035,.10,.17],trunk,[0,.12,0])
    for i,angle in enumerate([-2.25,-.3,1.35,2.85]):
        anchor=np.array([-.12,h*.49,.10]);spread=1.55+.22*(i%3)
        middle=anchor+np.array([.60*math.cos(angle),h*.14,.60*math.sin(angle)])
        end=np.array([spread*math.cos(angle),h*(.76+.025*i),spread*math.sin(angle)])
        bough=tube('Bough_'+str(i),[anchor.tolist(),middle.tolist(),end.tolist()],[.14,.10,.047],trunk,anchor.tolist())
        for j,sign in enumerate([-1,0,1]):
            t=.67 if j!=1 else .94;start=middle*(1-t)+end*t;direction=angle+sign*.67
            length=.77+.10*((i+j)%3);tip=start+np.array([length*math.cos(direction),h*(.055 if sign else .095),length*math.sin(direction)])
            tip[1]=min(tip[1],h*.92)
            shoot=tube('Shoot_'+str(i)+'_'+str(j),[start.tolist(),((start+tip)*.5+np.array([0,.08,0])).tolist(),tip.tolist()],[.044,.024,.009],bough,start.tolist(),8)
            # Each terminal spray is connected by closed petioles. Leaf groups
            # leave deliberate gaps instead of using spherical/alpha masses.
            for k in range(6):
                a=direction+(k-2.5)*.33;length=.36+.055*((i+j+k)%4);width=.11+.016*(k%3)
                stem=tip+np.array([0,-.03*(k%2),0]);leaf_base=stem+np.array([.11*math.cos(a),.015,.11*math.sin(a)])
                petiole=tube(f'Petiole_{i}_{j}_{k}',[stem.tolist(),leaf_base.tolist()],[.008,.0035],shoot,stem.tolist(),6)
                forward=np.array([math.cos(a),.10,math.sin(a)]);forward/=np.linalg.norm(forward)
                sideways=np.array([-math.sin(a),0,math.cos(a)]);up=np.array([0,1,0])
                local=[leaf_base,leaf_base+forward*length*.30-sideways*width*.82,
                       leaf_base+forward*length*.68-sideways*width,
                       leaf_base+forward*length,
                       leaf_base+forward*length*.68+sideways*width,
                       leaf_base+forward*length*.30+sideways*width*.82,
                       leaf_base+forward*length*.32+up*.040,
                       leaf_base+forward*length*.69+up*.030]
                faces=[[0,1,6],[1,2,6],[2,7,6],[2,3,7],[3,4,7],[4,5,7],[5,6,7],[5,0,6]]
                uv=[[.5,0],[0,.3],[0,.68],[.5,1],[1,.68],[1,.3],[.5,.32],[.5,.69]]
                part={'name':name+f'_Blade_{i}_{j}_{k}','vertices_world_xyz':[world(q)for q in local],'faces':faces,'role':['jade','sage','olive'][(i+j+k)%3],'category':'visual_foliage','cell':cell,'volume_m3':0,'open_foliage':True,'new_tree_source':True,'parent_path':petiole,'uv_per_vertex':uv}
                parts.append(part);attachments.append({'child':part['name'],'parent':petiole,'shared_or_embedded_joint_world_xyz':world(leaf_base),'construction_joint_only':True})
    own=[q for q in parts if q['name'].startswith(name+'_')]
    points=np.array([v for q in own for v in q['vertices_world_xyz']]);assert points[:,1].max()<base[1]+h+.12
    tree['actual_source_bounds_world_xyz']=[points.min(0).tolist(),points.max(0).tolist()]
(P/'docs/ARCHITECTURE_PARTS.json').write_text(json.dumps(data,separators=(',',':'))+'\n')
(P/'docs/BOTANICAL_PLAN.json').write_text(json.dumps(plan,indent=2)+'\n')
(P/'docs/ORIGINAL_TREE_HIERARCHY.json').write_text(json.dumps({'status':'ORIGINAL_SOURCE_HIERARCHY_AUTHORED_IN_NUMERIC_PLAN','trees':plan['procedural_trees'],'explicit_parent_joints':attachments,'source_existing_trees_scaled_or_changed':False,'sphere_clouds_or_alpha_cards':False,'biology_growth_wind_capacity_final_art_accepted':False},indent=2)+'\n')
print('G7_R007_ORIGINAL_TREE_PARTS',len(parts)-326,'parts',len(attachments),'parent joints',len(plan['procedural_trees']),'original medium trees')
