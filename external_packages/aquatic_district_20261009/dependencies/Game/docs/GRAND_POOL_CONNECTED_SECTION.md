## Reproduce the source-fit boundary

Run the following read-only check from the project root. It uses only the Python standard library, checks current source hashes, clips compiled triangles against each selected allocation and the extended staff shaft, and reproduces the principal area/ramp arithmetic. It does not launch Godot or Blender. The JSON separately records the raw GLB wall probes and allocation/hull checks; those are limited geometric evidence, not a substitute for future engine tests.

```python
import hashlib, json
from pathlib import Path
r = Path('.')
p = json.loads((r / 'data/architecture/GRAND_POOL_SECTION_PLAN.json').read_text())
for name, digest in p['source_sha256'].items():
    assert hashlib.sha256((r / name).read_bytes()).hexdigest() == digest, name
c = Path(p['catalogue_source_root'])
for name, digest in p['catalogue_source_sha256'].items():
    assert hashlib.sha256((c / name).read_bytes()).hexdigest() == digest, name

def area(poly):
    return abs(sum(a[0]*b[1]-b[0]*a[1]
        for a,b in zip(poly,poly[1:]+poly[:1]))) / 2

def clip(poly, rect):
    for axis, limit, sign in ((0,rect[0],1),(0,rect[1],-1),
                              (1,rect[2],1),(1,rect[3],-1)):
        out = []
        for a,b in zip(poly,poly[1:]+poly[:1]):
            ia = sign*(a[axis]-limit) >= -1e-10
            ib = sign*(b[axis]-limit) >= -1e-10
            if ia: out.append(a)
            if ia != ib:
                t = (limit-a[axis])/(b[axis]-a[axis])
                out.append([a[k]+t*(b[k]-a[k]) for k in (0,1)])
        poly = out
        if not poly: break
    return poly

s = json.loads((r / 'data/generated/structural_meshes.json').read_text())
assert len(s['regions']) == 250
assert s['source_checks']['source_sha256'] == p['source_sha256']['data/spatial_layout.json']
for target in p['programmes']:
    deck = target['key'][:3]
    overlap = sum(area(clip(t,target['allocation'])) for f in s['regions']
                  if f['deck'] == deck for t in f['triangles'])
    assert overlap < 1e-6, (deck,overlap)
shaft = p['staff']['shaft_reserved_rect']
for f in s['regions']:
    if f['floor_y'] > 15.1 and f['floor_y']-f['thickness'] < 55.74:
        assert sum(area(clip(t,shaft)) for t in f['triangles']) < 1e-6, f['id']
opening = area(p['shared_volume']['candidate_boundary_xz'])
extra = sum((b-a)*(d-c) for a,b,c,d in p['wet_market_link']['D09_opening_extensions_rects'])
assert abs(opening-1490) < 1e-6
assert abs(3128-opening-extra-1543.87) < 1e-6
assert abs(2*13.2/12-2.2) < 1e-6
print('Current-source hashes, zero selected compiled-floor overlap, shaft and dimensional checks pass')
```
